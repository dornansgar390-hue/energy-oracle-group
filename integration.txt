def fetch_pjm_dataset_lmp(api_key: str) -> float | None:
    url = "https://gridstatus.io"
    headers = {"x-api-key": api_key, "Content-Type": "application/json"}
    payload = {
        "filter": {
            "pnode_name": "PJM WEST HUB" # Строгая привязка к главному арбитражному хабу
        },
        "sort": [{"column": "timestamp", "direction": "desc"}],
        "limit": 1
    }
    try:
        res = requests.post(url, json=payload, headers=headers, timeout=10)
        if res.status_code == 200:
            # Извлекаем lmp цену из первой (самой свежей) строки датасета
            return float(res.json()["data"][0]["lmp"])
    except Exception as e:
        print(f"[WARN] Datasets API Error (PJM): {e}")
    return None

def fetch_interface_dataset_lmp(api_key: str, market: str, interface_pnode: str) -> float | None:
    """
    market: 'pjm', 'miso' или 'spp'
    interface_pnode: Имя пограничного узла (например, 'MISO_INTERFACE' или 'PJM_INTERFACE')
    """
    url = f"https://api.gridstatus.io/v1/datasets/{market}_rt_lmp_hourly/query"
    headers = {"x-api-key": api_key, "Content-Type": "application/json"}
    payload = {
        "filter": {
            "pnode_name": interface_pnode
        },
        "sort": [{"column": "timestamp", "direction": "desc"}],
        "limit": 1
    }
    try:
        res = requests.post(url, json=payload, headers=headers, timeout=10)
        if res.status_code == 200:
            return float(res.json()["data"][0]["lmp"])
    except Exception as e:
        print(f"[WARN] Interface Datasets API Error ({market}): {e}")
    return None
