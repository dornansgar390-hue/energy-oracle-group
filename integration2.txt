def fetch_pjm_cross_validation(api_key: str) -> float | None:
    """Запрашивает цену перетока PJM на стороне MISO для проверки PJM West Hub"""
    url = "https://gridstatus.io"
    headers = {"x-api-key": api_key, "Content-Type": "application/json"}
    payload = {
        "filter": {
            "pnode_name": "MISO.PJM_INTERFACE" # Фильтруем строго по интерфейсу
        },
        "sort": [{"column": "timestamp", "direction": "desc"}],
        "limit": 1
    }
    try:
        res = requests.post(url, json=payload, headers=headers, timeout=10)
        if res.status_code == 200:
            return float(res.json()["data"][0]["lmp"]) # Datasets API возвращает массив строк в ключе data
    except Exception as e:
        print(f"[WARN] Ошибка интерфейса MISO-PJM: {e}")
    return None
